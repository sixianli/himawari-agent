import { createServer } from "node:net";
import { expect, it } from "vitest";

const server = createServer();

it("listens on an ephemeral port", async () => {
  await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
  expect(server.address()).toMatchObject({ port: expect.any(Number) });
});

it("accepts an ephemeral option port and asserts a dial port", async () => {
  await new Promise((resolve) => server.listen({ host: "127.0.0.1", port: 0 }, resolve));
  expect({ host: "example.com", port: 443 }).toMatchObject({ port: 443 });
});
